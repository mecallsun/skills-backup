function onMeDormMonthChange() {
    var dormCode = document.getElementById('me_dorm').value;
    var readMonth = document.getElementById('me_month').value;
    if (!dormCode || !readMonth) return;

    fetch('/Meter/Index?handler=LoadReadings&dormCode=' + encodeURIComponent(dormCode) + '&readMonth=' + encodeURIComponent(readMonth))
        .then(r => r.json())
        .then(data => {
            if (!data.success) return;
            mePrevCold = data.prevCold;
            mePrevHot = data.prevHot;
            mePrevElectric = data.prevElectric;
            meHasPrev = data.hasPrev;
            meCurrentStatus = data.currentStatus;
            meCurrentRecordId = data.currentRecordId;

            // 显示重复警告（status=1/2 已生效 → 黄色警告；status=0 占位 → 蓝色提示）
            var dupWarn = document.getElementById('me_dupWarn');
            if (data.currentStatus >= 0) {
                var statusNames = ['未完成', '正常', '已修正', '未完成(PDA)', '已作废'];
                var statusName = statusNames[data.currentStatus] || '未知';
                if (data.isEffective) {
                    dupWarn.className = 'alert alert-warning mt-2 mb-0 py-1';

### 3.4 后端 LoadReadings Handler（Index.cshtml.cs 新增）

```csharp
/// <summary>
/// v2.13.81 新增：手动补录 Modal 的「加载上月读数 + 检测已存在记录」AJAX 端点
/// JS 在宿舍/月份变更时调用，返回 JSON 包含 prevCold/Hot/Electric、hasPrev、currentStatus 等
/// </summary>
public JsonResult OnGetLoadReadings([FromQuery] string? dormCode, [FromQuery] string? readMonth)
{
    if (string.IsNullOrWhiteSpace(dormCode) || string.IsNullOrWhiteSpace(readMonth))
        return new JsonResult(new { success = false, message = "参数不完整" });

    var dormCodeTrim = dormCode.Trim();
    var readMonthTrim = readMonth.Trim();

    var current = _db.MeterRecords
        .Where(r => r.DormCode == dormCodeTrim && r.ReadMonth == readMonthTrim)
        .OrderByDescending(r => r.ServerCreatedAt)
        .FirstOrDefault();

    var prev = _db.MeterRecords
        .Where(r => r.DormCode == dormCodeTrim && r.ReadMonth.CompareTo(readMonthTrim) < 0)
        .OrderByDescending(r => r.ReadMonth)
        .FirstOrDefault();

    var hasPrev = prev != null;
    var isEffective = current != null && (current.Status == 1 || current.Status == 2);

    return new JsonResult(new
    {
        success = true,
        prevCold = prev?.ColdMeter ?? 0m,
        prevHot = prev?.HotMeter ?? 0m,
        prevElectric = prev?.ElectricMeter ?? 0m,
        prevReadMonth = prev?.ReadMonth ?? "",
        hasPrev,
        currentStatus = current?.Status ?? -1,
        currentRecordId = current?.Id ?? 0L,
        isEffective
    });
}
```

## 六、关键决策

### 6.1 「弹窗 vs 独立页面」识别规则

用户在 v2.13.81 明确指出**所有类似的原型定义规则**：

| 原型 HTML 模式 | 识别 | 实现 |
|----------------|------|------|
| `<button data-bs-toggle="modal" data-bs-target="#xxx">` | **弹窗独占模式** | `OnClick="showXxxModal()"` + `#xxx` Modal |
| `<a href="xxx.html">` | **独立页面跳转** | `Url="/Xxx"` PageAction |
| `<button onclick="xxx()">` | **JS 函数** | 视场景：`OnClick` 或 `Url`（直连 API） |
| `<a href="xxx.html" target="_blank">` | **新窗口打开** | `Url="/Xxx" target="_blank"` |

**Meter Index 页判断结果**：
- 「手动补录」= Modal（`data-bs-toggle="modal"`）→ **弹窗独占**
- 「批量导入」= `<a href="import.html">` → **独立页面**
- 「导出Excel」= `<button onclick="exportExcel()">` → **直连后端 API**（更高效）
- 「新增记录」= 原型未提供 → **移除**

### 6.2 `data-bs-backdrop="static"` + `data-bs-keyboard="false"` 组合

弹窗独占模式的两道保险：
- `data-bs-backdrop="static"`：拦截 backdrop 点击关闭（防止误触丢失已录入数据）
- `data-bs-keyboard="false"`：拦截 ESC 键关闭（同上）
- 「取消」按钮（`data-bs-dismiss="modal"`）作为唯一主动关闭入口

### 6.3 LoadReadings vs 后端 OnPostSaveReadingsAsync

**避免职责重叠**：
| 场景 | 入口 | 调用端点 |
|------|------|----------|
| 弹窗中切换宿舍/月份（**仅查询**）| `onMeDormMonthChange()` | `GET /Meter?handler=LoadReadings` |
| 弹窗中提交保存 | `meSubmit()` | `POST /api/meter/manual-entry`（v2.13.80 已修复） |
| 独立 Entry 页面提交 | OnPostSaveReadingsAsync | 同上 manual-entry 端点 |

## 七、核心教训

1. **「原型交互模式」识别是 UI 1:1 对齐的前提**：用户在 v2.13.81 两次纠正后明确「所有类似的原型定义规则」——必须从原型 HTML 的 `data-bs-toggle="modal"` / `<a href="xxx.html">` / `onclick="xxx()"` 三种模式精准识别，再映射到 Razor 实现。
2. **「弹窗独占模式」= `data-bs-backdrop="static" data-bs-keyboard="false"` 组合**：与「可关闭 Modal」的 `backdrop=true` / `keyboard=true` 形成对照；弹窗独占用于「有未保存数据」场景，避免误触丢失。
3. **`PageHeader` PageAction 的 `Url` vs `OnClick` 选择**：`Url` 用于跳转（`<a href>`）或直连 API（`<a href="download.xlsx">`）；`OnClick` 用于调起 JS 函数（含 Modal）；两者都在前端渲染为 `<a>` 或 `<button>`。
4. **AJAX 端点命名规范**：Razor Pages `OnGet<X>` / `OnPost<X>` 方法通过 `?handler=X` 暴露为 AJAX 端点；本版本新增 `OnGetLoadReadings` → `?handler=LoadReadings`。
5. **历史数据检测的副作用**：本次端到端测试时发现 v2.13.79 历史的 A101 2026-08 记录（id=3, status=1）会被 `LoadReadings` 检测到 → 黄色警告，强制要求走修正流程——这是 v2.11.4 业务规则在 v2.13.81 弹窗独占模式下的正确实现。
