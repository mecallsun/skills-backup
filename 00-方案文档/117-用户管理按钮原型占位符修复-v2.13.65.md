```js
    if (!confirm(`确定重启 ${which === 'pda' ? 'API' : 'Web Admin'} 服务吗？`)) return;

        }
    })
    .catch(err => alert('请求失败：' + err.message));
}
```

### 3.2 User.cshtml 全面 AJAX 化

**核心改造**：
- 移除 `asp-page-handler="..."` 表单提交（避免整页跳转）
- 改用 `<form id="...">` 纯 HTML 表单 + JS 监听 submit 事件
- 改用 Bootstrap 5 原生 `data-bs-toggle="modal" data-bs-target="..."` 触发（不再依赖内联 `onclick`）
- 数据通过 `data-user-id` / `data-display-name` / `data-role-ids` 等 `data-*` 属性传递到 modal
- 提交按钮带 loading spinner + 提交期间禁用

**关键代码**：
```js
// PageHandler 通过 form.submit 改为 JS fetch
document.getElementById('createUserForm').addEventListener('submit', async function(e) {
    e.preventDefault();
    hideFeedback('createUserFeedback');
    const data = new FormData(this);
    const submitBtn = document.getElementById('createUserSubmit');
    const origHtml = submitBtn.innerHTML;
    submitBtn.disabled = true;
    submitBtn.innerHTML = '<span class="spinner-border spinner-border-sm me-1"></span>保存中...';
    const result = await postUserAction(
        '/Settings/User?handler=Create',
        data,
        'createUserFeedback'
    );
    submitBtn.disabled = false;
    submitBtn.innerHTML = origHtml;
    if (result.ok) {
        showFeedback('createUserFeedback', 'success', '✓ 创建成功！即将刷新...');
        setTimeout(() => location.reload(), 800);
    }
    // 失败时 postUserAction 内部已显示 feedback
});
```

### 3.3 页面级 Antiforgery Token 注入

为 AJAX 表单提交添加页面级 antiforgery token：

```html
@inject Microsoft.AspNetCore.Antiforgery.IAntiforgery PageAntiforgery
@{
    var pageTokens = PageAntiforgery.GetAndStoreTokens(HttpContext);
}
<input type="hidden" id="pageAntiforgeryToken" name="__RequestVerificationToken" value="@pageTokens.RequestToken" />
```

JS 读取后通过 `RequestVerificationToken` header 提交：
```js
const antiforgeryToken = document.querySelector('#pageAntiforgeryToken')?.value || '';
fetch(url, {
    method: 'POST',
    headers: {
        'Content-Type': 'application/x-www-form-urlencoded',
        'RequestVerificationToken': antiforgeryToken
    },
    body: formData.toString()
});
```

### 3.4 PageModel handler 返回 JSON 而非 Redirect

```csharp
// 旧（v2.13.46 RedirectToPage 模式）
public async Task<IActionResult> OnPostCreateAsync(...) {
    ...
    TempData["Success"] = "创建成功";
    return RedirectToPage();  // ← 302 跳转，AJAX 无法处理
}

// 新（v2.13.65 JSON 模式）
public async Task<IActionResult> OnPostCreateAsync(...) {
    ...
    return new JsonResult(new { success = true, message = "创建成功", userId = user.Id });
}
```

### 3.5 删除操作改用 ExecuteDeleteAsync（修复 DbUpdateConcurrencyException）

```csharp
// 旧（Remove + SaveChangesAsync 会触发 EF 并发检查）
var urs = _db.SysUserRoles.Where(ur => ur.UserId == id);
_db.SysUserRoles.RemoveRange(urs);
...
await _db.SaveChangesAsync();
// → DbUpdateConcurrencyException: expected 1, affected 0

// 新（ExecuteDeleteAsync 直发 SQL DELETE，无并发检查）
await _db.SysUserRoles.Where(ur => ur.UserId == id).ExecuteDeleteAsync();
await _db.SysUserSecurityQuestions.Where(sq => sq.UserId == id).ExecuteDeleteAsync();
await _db.SysOpLogs.Where(l => l.UserId == id).ExecuteDeleteAsync();
await _db.SysUserFilterCaches.Where(c => c.UserId == id).ExecuteDeleteAsync();
await _db.SysUsers.Where(u => u.Id == id).ExecuteDeleteAsync();
```

**根因**：EF Core 在 `Remove` + `SaveChangesAsync` 时携带**所有属性的原值**生成 WHERE 子句；任一属性与 DB 不一致（并发或 race condition）→ WHERE 不匹配 → 0 rows affected → 抛 `DbUpdateConcurrencyException`。`ExecuteDeleteAsync` 是 EF Core 7+ 提供的「直发 DELETE」方法，绕过 ChangeTracker，无并发检查。

## 七、教训总结

### 7.1 三大关键教训

1. ❌ **「弹 alert 是原型」= 占位符遗留 = P0 必修** — 如果用户能复现「原型演示」字样，说明仍有未替换的占位代码；必须全局 grep + 替换
2. ❌ **AJAX 化时一定要注入 antiforgery token** — `asp-page-handler` 自动注入的 hidden field 不会出现在纯 HTML form 中；必须用 `@inject Antiforgery` 手动添加
3. ❌ **删除操作慎用 `Remove + SaveChangesAsync`** — EF Core 会做并发检查，0 rows affected 抛异常；用 `ExecuteDeleteAsync` 直发 SQL 更可靠

### 7.2 强制检查清单（CRUD 按钮自检）

- [ ] 全局 grep `原型演示` — 必须 0 命中（除文档/注释）
- [ ] AJAX form 必须有 antiforgery token
- [ ] PageModel handler 返回 `JsonResult` 而非 `RedirectToPage`
- [ ] 删除操作优先用 `ExecuteDeleteAsync`
- [ ] 按钮要有可见的 loading 状态 + 成功/失败 feedback
- [ ] 端到端 curl 测试 4 个 CRUD 操作全部 200
